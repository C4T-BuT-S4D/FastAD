Rating Formula
// FastAD rating delta calculation
const scale = 50 * Math.sqrt(hardness);
const norm = Math.log(Math.log(hardness)) / 12;
const ratingDelta = Math.sqrt(attackerScore) - Math.sqrt(victimScore);
const ratingDeltaNorm = ratingDelta * norm;
const attackerDelta = scale / (1 + Math.exp(ratingDeltaNorm));
